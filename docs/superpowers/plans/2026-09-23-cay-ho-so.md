```ts
export interface DossierWorkspaceState {
  activeDossierId: string;
  byDossier: Record<string, DossierState>;
}

export type WorkspaceAction =
  | DossierAction
  | { type: 'selectDossier'; dossierId: string }
  | { type: 'selectDocIn'; dossierId: string; docId: string }
  | { type: 'toggleOptionalIn'; dossierId: string; docId: string };

export function initWorkspaceState(dossierId: string): DossierWorkspaceState {
  return { activeDossierId: dossierId, byDossier: { [dossierId]: initDossierState(dossierId) } };
}

function withDossier(state: DossierWorkspaceState, dossierId: string, action?: DossierAction): DossierWorkspaceState {
  const current = state.byDossier[dossierId] ?? initDossierState(dossierId);
  const next = action ? dossierReducer(current, action) : current;
  return { ...state, byDossier: { ...state.byDossier, [dossierId]: next } };
}

export function workspaceReducer(state: DossierWorkspaceState, action: WorkspaceAction): DossierWorkspaceState {
  switch (action.type) {
    case 'selectDossier':
      return { ...withDossier(state, action.dossierId), activeDossierId: action.dossierId };
    case 'selectDocIn':
      return {
        ...withDossier(state, action.dossierId, { type: 'selectDoc', docId: action.docId }),
        activeDossierId: action.dossierId,
      };
    case 'toggleOptionalIn':
      return withDossier(state, action.dossierId, { type: 'toggleOptional', docId: action.docId });
    default:
      return withDossier(state, state.activeDossierId, action);
  }
}
```

```ts
import type { ComponentType } from 'react';

export interface TreeNode {
  id: string;
  label: string;
  icon?: ComponentType<{ className?: string; size?: number }>;
  children?: TreeNode[];
  optional?: boolean;
  disabled?: boolean;
}

export function pathTo(nodes: TreeNode[], id: string): string[] {
  for (const n of nodes) {
    if (n.id === id) return [];
    if (n.children) {
      const sub = pathTo(n.children, id);
      if (sub.length > 0 || n.children.some(c => c.id === id)) return [n.id, ...sub];
    }
  }
  return [];
}
```

```ts
import type { ComponentType } from 'react';
import type { TreeNode } from '../tree/treeNodes';

export interface DossierLike {
  id: string;
  label: string;
  icon?: ComponentType<{ className?: string; size?: number }>;
  documents: Array<{ templateId: string; optional?: boolean }>;
}

export const nodeIdOf = (dossierId: string, docId: string) => `${dossierId}/${docId}`;

export function parseNodeId(id: string): { dossierId: string; docId: string } | null {
  const i = id.indexOf('/');
  if (i <= 0 || i === id.length - 1) return null;
  return { dossierId: id.slice(0, i), docId: id.slice(i + 1) };
}

export function buildDossierTree(
  dossiers: DossierLike[],
  labelOf: (docId: string) => string,
  isExcludedIn: (dossierId: string, docId: string) => boolean,
): TreeNode[] {
  return dossiers.map(d => ({
    id: d.id,
    label: d.label,
    icon: d.icon,
    children: d.documents.map(doc => ({
      id: nodeIdOf(d.id, doc.templateId),
      label: labelOf(doc.templateId),
      optional: doc.optional === true,
      disabled: isExcludedIn(d.id, doc.templateId),
    })),
  }));
}
```

```ts
import { useCallback, useState } from 'react';

function read(key: string, initial: boolean): boolean {
  try {
    const raw = window.localStorage.getItem(key);
    return raw === null ? initial : raw === '1';
  } catch {
    return initial;
  }
}

export function usePersistedFlag(key: string, initial: boolean): [boolean, (v: boolean) => void] {
  const [value, setValue] = useState(() => read(key, initial));
  const set = useCallback((v: boolean) => {
    setValue(v);
    try {
      window.localStorage.setItem(key, v ? '1' : '0');
    } catch {
      return;
    }
  }, [key]);
  return [value, set];
}
```

```tsx
import React, { useState } from 'react';
import { ChevronDown, ChevronRight, PanelLeftClose } from 'lucide-react';
import { pathTo, type TreeNode } from '../../lib/tree/treeNodes';

interface TreeRailProps {
  title: string;
  nodes: TreeNode[];
  selectedId: string;
  onSelect: (id: string) => void;
  onCollapse: () => void;
  renderTrailing?: (node: TreeNode) => React.ReactNode;
  className?: string;
}

export const TreeRail: React.FC<TreeRailProps> = ({ title, nodes, selectedId, onSelect, onCollapse, renderTrailing, className = '' }) => {
  const [open, setOpen] = useState<Set<string>>(() => new Set());
  const forced = new Set(pathTo(nodes, selectedId));
  const isOpen = (id: string) => forced.has(id) || open.has(id);
  const toggle = (id: string) => setOpen(prev => {
    const next = new Set(prev);
    if (next.has(id)) next.delete(id);
    else next.add(id);
    return next;
  });

  const renderNodes = (list: TreeNode[], depth: number): React.ReactNode => (
    <ul className="space-y-0.5">
      {list.map(node => {
        const Icon = node.icon;
        if (node.children) {
          const expanded = isOpen(node.id);
          return (
            <li key={node.id}>
              <button
                type="button"
                aria-expanded={expanded}
                onClick={() => toggle(node.id)}
                style={{ paddingLeft: `${8 + depth * 12}px` }}
                className="w-full flex items-center gap-1.5 pr-2 py-1.5 text-xs font-semibold text-ink hover:bg-panel-muted/60 transition-colors text-left"
              >
                {expanded ? <ChevronDown size={13} className="text-ink-faint shrink-0" aria-hidden /> : <ChevronRight size={13} className="text-ink-faint shrink-0" aria-hidden />}
                {Icon && <Icon size={14} className="text-ink-muted shrink-0" />}
                <span className="truncate">{node.label}</span>
              </button>
              {expanded && renderNodes(node.children, depth + 1)}
            </li>
          );
        }
        const selected = node.id === selectedId;
        return (
          <li key={node.id} className="flex items-center">
            <button
              type="button"
              disabled={node.disabled}
              aria-current={selected ? 'true' : undefined}
              onClick={() => onSelect(node.id)}
              title={node.label}
              style={{ paddingLeft: `${8 + depth * 12}px` }}
              className={`flex-1 min-w-0 text-left pr-2 py-1.5 text-xs border-l-2 transition-colors truncate ${
                selected
                  ? 'border-accent bg-panel-muted font-bold text-ink'
                  : node.disabled
                  ? 'border-transparent text-ink-faint line-through opacity-50 cursor-not-allowed'
                  : 'border-transparent text-ink-muted hover:text-ink hover:bg-panel-muted/60'
              }`}
            >
              {node.label}
            </button>
            {renderTrailing?.(node)}
          </li>
        );
      })}
    </ul>
  );

  return (
    <nav aria-label={title} className={`flex flex-col bg-panel text-ink ${className}`}>
      <div className="flex items-center justify-between px-3 py-2 border-b border-line/60">
        <span className="text-[11px] font-bold uppercase tracking-wide text-ink-faint">{title}</span>
        <button
          type="button"
          onClick={onCollapse}
          aria-label={`Ẩn ${title}`}
          title={`Ẩn ${title}`}
          className="p-1 rounded text-ink-muted hover:text-ink hover:bg-panel-muted/60"
        >
          <PanelLeftClose size={15} />
        </button>
      </div>
      <div className="flex-1 overflow-y-auto py-1.5">{renderNodes(nodes, 0)}</div>
    </nav>
  );
};
```

`DossierTreeRail.tsx`:

```tsx
import React, { useMemo } from 'react';
import type { useDocumentDossierController } from '../../../hooks/useDocumentDossierController';
import { TEMPLATE_BY_ID } from '../registry/templates';
import { TreeRail } from '../../common/TreeRail';
import { buildDossierTree, nodeIdOf, parseNodeId } from '../../../lib/documents/dossierTree';

type Ctrl = ReturnType<typeof useDocumentDossierController>;

export const DossierTreeRail: React.FC<{ ctrl: Ctrl; onCollapse: () => void; onPicked?: () => void; className?: string }> = ({ ctrl, onCollapse, onPicked, className }) => {
  const nodes = useMemo(
    () => buildDossierTree(ctrl.dossiers, id => TEMPLATE_BY_ID[id]?.label ?? id, ctrl.isExcludedIn),
    [ctrl.dossiers, ctrl.isExcludedIn],
  );
  const labelOfDossier = (id: string) => ctrl.dossiers.find(d => d.id === id)?.label ?? id;

  return (
    <TreeRail
      title="Bộ hồ sơ"
      nodes={nodes}
      selectedId={nodeIdOf(ctrl.state.activeDossierId, ctrl.state.activeDocId)}
      onSelect={id => {
        const p = parseNodeId(id);
        if (!p) return;
        ctrl.selectDocIn(p.dossierId, p.docId);
        onPicked?.();
      }}
      onCollapse={onCollapse}
      className={className}
      renderTrailing={node => {
        if (!node.optional) return null;
        const p = parseNodeId(node.id);
        if (!p) return null;
        return (
          <label className="flex items-center gap-1 px-2 text-[10px] text-ink-faint cursor-pointer select-none shrink-0">
            <input
              type="checkbox"
              checked={!ctrl.isExcludedIn(p.dossierId, p.docId)}
              onChange={() => ctrl.toggleOptionalIn(p.dossierId, p.docId)}
              aria-label={`Gồm ${node.label} (${labelOfDossier(p.dossierId)})`}
              className="rounded border-line w-3 h-3"
            />
            gồm
          </label>
        );
      }}
    />
  );
};
```
